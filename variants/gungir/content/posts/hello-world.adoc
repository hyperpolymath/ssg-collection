=== Code Example

Here’s a simple example in ReScript:

[source,rescript]
----
// Simple greeting
Js.log("Hello, World!")

// Pattern matching example
type page = Index | About | Post(string)

let renderTitle = (page: page): string => {
  switch page {
  | Index => "Welcome"
  | About => "About Us"
  | Post(title) => title
  }
}

// Pipe-first syntax
["index", "about", "contact"]
->Array.map(name => name->String.toUpperCase)
->Array.forEach(Js.log)
----
